# Single-container pod image. Phase 2 installs Postgres 17, pgvector, s6-overlay,
# cloudflared, and tinyproxy on top of the cuda app image and runs /init.
# This file is the boot contract those layers call: lease, socket-only Postgres,
# then ndup db migrate. It builds without a RunPod account.

FROM python:3.12-slim

ENV PYTHONDONTWRITEBYTECODE=1 \
    PYTHONUNBUFFERED=1 \
    HF_HUB_OFFLINE=1 \
    PGDATA=/workspace/pgdata \
    NDUP_LEASE=/workspace/ndup.lease

WORKDIR /app
COPY pyproject.toml README.md uv.lock ./
COPY src ./src
RUN pip install --no-cache-dir .
COPY pod/entrypoint.sh /entrypoint.sh
RUN chmod +x /entrypoint.sh

VOLUME ["/workspace"]
ENTRYPOINT ["/entrypoint.sh"]
CMD ["ndup", "doctor"]
